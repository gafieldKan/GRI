/**
 * 支付宝支付策略
 */
class AlipayStrategy extends PaymentStrategy {
  async pay(amount, context) {
    // 调用支付宝 API

      provider: 'alipay',
      transactionId: `AL${Date.now()}`,
      amount,
      status: 'success'
    }
  }

/**
 * 支付工厂
 * 体现：工厂模式
 */
class PaymentFactory {
  static #strategies = {
    'wechat': WechatPayStrategy,
    'alipay': AlipayStrategy
  }
  
  /**
   * 创建支付策略实例
   * @param {string} type - 支付类型

  static create(type) {
    const StrategyClass = this.#strategies[type]
    if (!StrategyClass) {
      throw new Error(`不支持的支付类型：${type}`)
    }
    return new StrategyClass()
  }
  
  /**
   * 注册新的支付策略（扩展点）
   */
  static register(type, strategyClass) {
    this.#strategies[type] = strategyClass
  }
}

/**
 * 数据访问代理
 * 体现：代理模式 - 用于权限控制和缓存
 */
class DataAccessProxy {
  constructor(realSubject, accessControl) {
    this.realSubject = realSubject
    this.accessControl = accessControl
    this.cache = new Map()
  }
  
  async getData(id, user) {
    // 前置处理：权限校验
    this.#checkPermission(user, 'read')
    
    // 缓存逻辑
    if (this.cache.has(id)) {
      return this.cache.get(id)
    }
    
    // 调用真实对象
    const data = await this.realSubject.getData(id)
    this.cache.set(id, data)
    return data
  }
  
  async updateData(id, data, user) {
    // 权限校验
    this.#checkPermission(user, 'write')
    
    // 调用真实对象
    const result = await this.realSubject.updateData(id, data)
    
    // 清除缓存
    this.cache.delete(id)
    
    return result
  }
  
  #checkPermission(user, action) {
    if (!this.accessControl.can(user, action)) {
      throw new Error('无权限')
    }
  }
}

class LoggerDecorator {
  constructor(service) {
    this.service = service
  }

    console.log('[LOG] 开始执行:', new Date().toISOString())
    console.log('[LOG] 输入数据:', data)

      console.log('[LOG] 耗时:', duration, 'ms')
      
      return result
    } catch (error) {
      console.error('[LOG] 执行失败:', error)
      throw error
    }
  }
}

/**
 * 事务装饰器
 */
class TransactionDecorator {
  constructor(service, transactionManager) {
    this.service = service
    this.transactionManager = transactionManager
  }
  
  async execute(data) {
    await this.transactionManager.begin()
    try {
      const result = await this.service.execute(data)
      await this.transactionManager.commit()
      return result
    } catch (error) {
      await this.transactionManager.rollback()
      throw error
    }
  }
}

class OrderObservable {
  constructor() {
    this.observers = []
  }
  
  /**
   * 订阅
   */
  subscribe(observer) {
    this.observers.push(observer)
  }
  
  /**
   * 取消订阅
   */
  unsubscribe(observer) {

  }

   * 通知所有观察者
   */
  notify(event, data) {
    this.observers.forEach(observer => {
      observer.update(event, data)
    })
  }
}

    this.smsService = smsService

    switch (event) {
      case 'ORDER_PAID':
        this.emailService.sendPaidNotice(data)
        break
      case 'ORDER_SHIPPED':
        this.smsService.sendShippedNotice(data)
        break
      case 'ORDER_COMPLETED':
        this.emailService.sendCompletionNotice(data)
        break
    }
  }

## 三、钩子设计示例

### 3.1 完整的钩子系统

```javascript
/**
 * 钩子管理器
 */
class HookManager {
  constructor() {
    this.hooks = new Map()
  }
  
  /**
   * 注册钩子
   */
  register(name, callback, priority = 0) {
    if (!this.hooks.has(name)) {
      this.hooks.set(name, [])
    }
    const hooks = this.hooks.get(name)
    hooks.push({ callback, priority })
    // 按优先级排序
    hooks.sort((a, b) => a.priority - b.priority)
  }
  
  /**
   * 触发钩子
   */
  async trigger(name, context = {}) {
    const hooks = this.hooks.get(name)
    if (!hooks) return
    
    for (const { callback } of hooks) {
      await callback(context)
    }
  }
}

/**
 * 可挂载钩子的实体基类
 */
class HookableEntity {
  constructor() {
    this.hookManager = new HookManager()
    this.#registerDefaultHooks()
  }
  
  // 注册默认钩子
  #registerDefaultHooks() {
    const defaultHooks = [
      'beforeCreate', 'afterCreate',
      'beforeUpdate', 'afterUpdate',
      'beforeDelete', 'afterDelete',
      'beforeValidate', 'afterValidate',
      'beforeSave', 'afterSave'
    ]
    defaultHooks.forEach(name => {
      this.hookManager.register(name, () => {})
    })
  }
  
  // 添加钩子监听器
  on(name, callback, priority) {
    this.hookManager.register(name, callback, priority)
  }
  
  // 触发钩子
  async triggerHook(name, context) {
    await this.hookManager.trigger(name, context)
  }
}
```

/**
 * 配置服务
 * 体现：配置优先原则
 */
class ConfigService {
  constructor() {
    this.layers = {
      system: new Map(),    // 系统默认
      tenant: new Map(),    // 租户配置
      org: new Map(),       // 组织配置
      user: new Map()       // 用户配置
    }
  }
  
  /**
   * 获取配置值（分层合并）
   */
  get(key, defaultValue) {
    // 从各层查找，返回第一个非空值
    for (const layer of Object.values(this.layers)) {
      if (layer.has(key)) {
        return layer.get(key)
      }
    }
    return defaultValue
  }
  
  /**
   * 设置配置
   */
  set(layer, key, value) {
    if (this.layers[layer]) {
      this.layers[layer].set(key, value)
    }
  }
}

// 配置示例
const config = new ConfigService()

// 系统默认配置
config.layers.system.set('maxOrderAmount', 10000)
config.layers.system.set('allowedCurrencies', ['CNY', 'USD', 'EUR'])

// 租户配置（覆盖系统）
config.layers.tenant.set('maxOrderAmount', 50000)

// 获取配置（自动合并）

/**
 * 功能开关服务
 */
class FeatureService {
  constructor(configService) {
    this.config = configService
    this.features = new Map()
  }
  
  /**
   * 注册功能
   */
  register(name, config) {
    this.features.set(name, {
      enabled: config.enabled ?? true,
      description: config.description,
      conditions: config.conditions || [],
      metadata: config.metadata || {}
    })
  }
  
  /**
   * 检查功能是否可用
   */
  isEnabled(name, context = {}) {
    const feature = this.features.get(name)
    if (!feature) return false
    if (!feature.enabled) return false
    
    // 检查条件
    for (const condition of feature.conditions) {
      if (!this.#evaluateCondition(condition, context)) {
        return false
      }
    }
    
    return true
  }
  
  #evaluateCondition(condition, context) {
    const { field, operator, value } = condition
    
    switch (operator) {
      case 'in':
        return value.includes(context[field])
      case 'equals':
        return context[field] === value
      case 'greaterThan':
        return context[field] > value
      default:
        return true
    }
  }
}

// 使用示例
const featureService = new FeatureService(config)

// 注册多币种功能
featureService.register('multiCurrency', {
  enabled: true,
  description: '多币种支付支持',
  conditions: [
    { field: 'region', operator: 'in', value: ['CN', 'US', 'EU'] },
    { field: 'userLevel', operator: 'greaterThan', value: 1 }
  ]
})

// 检查功能

  region: 'CN', 
  userLevel: 2

}

class I18nService {
  constructor() {
    this.translations = new Map()
    this.currentLocale = 'zh-CN'
    this.fallbackLocale = 'en-US'
  }
  
  /**
   * 设置当前语言环境
   */
  setLocale(locale) {
    this.currentLocale = locale
  }
  
  /**
   * 翻译文本
   */
  t(key, params = {}) {
    const translation = this.translations.get(key)
    if (!translation) return key
    
    // 获取当前语言翻译，回退到默认语言
    let text = translation[this.currentLocale] || 
               translation[this.fallbackLocale] || 
               key
    
    // 参数替换
    Object.entries(params).forEach(([k, v]) => {
      text = text.replace(`{${k}}`, v)
    })
    
    return text
  }
  
  /**
   * 加载翻译文件
   */
  loadTranslations(locale, translations) {
    Object.entries(translations).forEach(([key, value]) => {
      if (!this.translations.has(key)) {
        this.translations.set(key, {})
      }
      this.translations.get(key)[locale] = value
    })
  }
}

// 使用示例
const i18n = new I18nService()

// 加载翻译
i18n.loadTranslations('zh-CN', {
  'order.status.pending': '待支付',
  'order.status.paid': '已支付',
  'order.amount': '订单金额：{amount}'
})

i18n.loadTranslations('en-US', {
  'order.status.pending': 'Pending',
  'order.status.paid': 'Paid',
  'order.amount': 'Order Amount: {amount}'
})

// 使用
console.log(i18n.t('order.status.pending')) // 待支付
console.log(i18n.t('order.amount', { amount: '100' })) // 订单金额：100
